cmake_minimum_required(VERSION 3.20)
project(fb_present_dump_regression C)
enable_testing()
if(NOT WIN32)
    message(FATAL_ERROR "This regression targets the Windows framebuffer presenter")
endif()
add_executable(fb-present-dump-test fb_present_dump_test.c ../../src/video/fb_present.c)
target_include_directories(fb-present-dump-test PRIVATE ../../src/input ../../src)
target_compile_definitions(fb-present-dump-test PRIVATE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(fb-present-dump-test PRIVATE user32 gdi32)
foreach(mode disabled rgb32 rgb565 aa2x1 aa2x2)
    add_test(NAME present-${mode} COMMAND fb-present-dump-test ${mode} "${CMAKE_CURRENT_BINARY_DIR}/${mode}.bmp")
endforeach()
